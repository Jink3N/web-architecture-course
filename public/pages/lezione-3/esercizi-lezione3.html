// public/app.js
const API_URL = 'http://localhost:3000/api/tasks';

// Carica tasks all'avvio
document.addEventListener('DOMContentLoaded', loadTasks);

// Carica e mostra tasks
async function loadTasks() {
    try {
        const response = await fetch(API_URL);
        const result = await response.json();
        
        if (result.success) {
            displayTasks(result.data);
        }
    } catch (error) {
        showMessage('Errore caricamento tasks', 'error');
    }
}

// Mostra tasks nella UI
function displayTasks(tasks) {
    const taskList = document.getElementById('taskList');
    
    if (tasks.length === 0) {
        taskList.innerHTML = '<p style="text-align: center; color: #999;">Nessun task presente</p>';
        return;
    }
    
    taskList.innerHTML = tasks.map(task => `
        <div class="task-item ${task.completed ? 'completed' : ''}">
            <span class="task-text">${task.description}</span>
            <div class="task-actions">
                ${!task.completed ? `
                    <button class="btn-complete" onclick="completeTask(${task.id})">✓</button>
                ` : ''}
                <button class="btn-delete" onclick="deleteTask(${task.id})">🗑️</button>
            </div>
        </div>
    `).join('');
}

// Aggiungi nuovo task
async function addTask() {
    const input = document.getElementById('taskInput');
    const description = input.value.trim();
    
    if (!description) {
        showMessage('Inserisci una descrizione', 'error');
        return;
    }
    
    try {
        const response = await fetch(API_URL, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ description })
        });
        
        const result = await response.json();
        
        if (result.success) {
            input.value = '';
            showMessage('Task aggiunto!', 'success');
            loadTasks();
        } else {
            showMessage(result.error, 'error');
        }
    } catch (error) {
        showMessage('Errore durante l\'aggiunta', 'error');
    }
}

// Completa task
async function completeTask(id) {
    try {
        const response = await fetch(`${API_URL}/${id}/complete`, {
            method: 'PATCH'
        });
        
        const result = await response.json();
        
        if (result.success) {
            showMessage('Task completato!', 'success');
            loadTasks();
        }
    } catch (error) {
        showMessage('Errore durante il completamento', 'error');
    }
}

// Elimina task
async function deleteTask(id) {
    if (!confirm('Sei sicuro di voler eliminare questo task?')) {
        return;
    }
    
    try {
        const response = await fetch(`${API_URL}/${id}`, {
            method: 'DELETE'
        });
        
        const result = await response.json();
        
        if (result.success) {
            showMessage('Task eliminato!', 'success');
            loadTasks();
        }
    } catch (error) {
        showMessage('Errore durante l\'eliminazione', 'error');
    }
}

// Mostra messaggio feedback
function showMessage(text, type) {
    const message = document.getElementById('message');
    message.textContent = text;
    message.className = `message ${type}`;
    
    setTimeout(() => {
        message.classList.add('hidden');
    }, 3000);
}

// Enter per aggiungere
document.getElementById('taskInput').addEventListener('keypress', (e) => {
    if (e.key === 'Enter') addTask();
});
