🔄 Stream & File System

Gestire file grandi e flussi di dati in modo efficiente

📂 File System (fs) Avanzato

Docs: Node.js fs 

import fs from 'fs/promises';
import path from 'path';

// Leggere file
const content = await fs.readFile('data.txt', 'utf-8');
const json = await fs.readFile('config.json', 'utf-8');
const data = JSON.parse(json);

// Scrivere file
await fs.writeFile('output.txt', 'Hello World');
await fs.writeFile('data.json', JSON.stringify(data, null, 2));

// Append (aggiungere alla fine)
await fs.appendFile('log.txt', 'New line\n');

// Copiare file
await fs.copyFile('source.txt', 'destination.txt');

// Rinominare/spostare
await fs.rename('old.txt', 'new.txt');

// Cancellare
await fs.unlink('file.txt');
await fs.rm('file.txt');  // Alias moderno

// Directory
await fs.mkdir('new-folder');
await fs.mkdir('nested/folders', { recursive: true });
const files = await fs.readdir('./');
await fs.rmdir('folder');
await fs.rm('folder', { recursive: true });

// Info file
const stats = await fs.stat('file.txt');
console.log(stats.size);          // Dimensione bytes
console.log(stats.isFile());      // true
console.log(stats.isDirectory()); // false
console.log(stats.mtime);         // Data modifica

// Controllare esistenza
try {
    await fs.access('file.txt');
    console.log('File esiste');
} catch {
    console.log('File non esiste');
}

🌊 Streams: la chiave per file grandi

Gli stream processano dati "a pezzi" invece di caricare tutto in memoria. Docs: Node.js Streams

Tipi di Stream

Tipo Descrizione Esempi
Readable Sorgente da leggere fs.createReadStream, http.IncomingMessage
Writable Destinazione dove scrivere fs.createWriteStream, http.ServerResponse
Duplex Entrambi (read + write) net.Socket, TCP sockets
Transform Modifica dati in transito zlib.createGzip, crypto streams

Leggere file con stream

import fs from 'fs';

// File grande (10GB) - solo pochi MB in memoria
const readStream = fs.createReadStream('huge-file.txt', {
    encoding: 'utf-8',
    highWaterMark: 64 * 1024  // 64KB chunks
});

readStream.on('data', (chunk) => {
    console.log('Ricevuto chunk:', chunk.length);
});

readStream.on('end', () => {
    console.log('Finito!');
});

readStream.on('error', (err) => {
    console.error('Errore:', err);
});

Scrivere file con stream

const writeStream = fs.createWriteStream('output.txt');

writeStream.write('Prima riga\n');
writeStream.write('Seconda riga\n');
writeStream.end('Ultima riga\n');

writeStream.on('finish', () => {
    console.log('Scrittura completata');
});

Pipe: collegare stream

// Copia file efficiente (nessun buffer in memoria)
const read = fs.createReadStream('input.txt');
const write = fs.createWriteStream('output.txt');
read.pipe(write);

// Compressione con pipeline
import { pipeline } from 'stream/promises';
import { createGzip } from 'zlib';

await pipeline(
    fs.createReadStream('large.txt'),
    createGzip(),
    fs.createWriteStream('large.txt.gz')
);

// Trasformazione custom
import { Transform } from 'stream';

const upperCaseTransform = new Transform({
    transform(chunk, encoding, callback) {
        this.push(chunk.toString().toUpperCase());
        callback();
    }
});

fs.createReadStream('input.txt')
    .pipe(upperCaseTransform)
    .pipe(fs.createWriteStream('output.txt'));

🚀 Casi d'uso reali

1. Upload file grandi

import express from 'express';

app.post('/upload', (req, res) => {
    const writeStream = fs.createWriteStream('uploaded-file.pdf');
    
    req.pipe(writeStream);
    
    writeStream.on('finish', () => {
        res.json({ success: true });
    });
    
    writeStream.on('error', (err) => {
        res.status(500).json({ error: err.message });
    });
});

2. Log file rotation

import fs from 'fs';

let logStream = fs.createWriteStream('app.log', { flags: 'a' });
let lineCount = 0;

function log(message) {
    logStream.write(`${new Date().toISOString()} ${message}\n`);
    lineCount++;
    
    // Rotazione ogni 1000 righe
    if (lineCount >= 1000) {
        logStream.end();
        fs.rename('app.log', `app-${Date.now()}.log`, () => {
            logStream = fs.createWriteStream('app.log', { flags: 'a' });
            lineCount = 0;
        });
    }
}

3. CSV Processing

import { createReadStream } from 'fs';
import { createInterface } from 'readline';

const rl = createInterface({
    input: createReadStream('data.csv'),
    crlfDelay: Infinity
});

let rowCount = 0;
for await (const line of rl) {
    const [name, age, city] = line.split(',');
    console.log({ name, age, city });
    rowCount++;
}
console.log(`Processate ${rowCount} righe`);

💡 Best Practices

  • ✅ Usa stream per file maggiori di 10MB
  • ✅ Sempre gestire eventi 'error'
  • ✅ Usa pipeline() invece di .pipe() per gestione errori
  • ✅ Chiudi sempre gli stream con .end()
  • ✅ Backpressure: rispetta write() che ritorna false
