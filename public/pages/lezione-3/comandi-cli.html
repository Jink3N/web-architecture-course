💻 Comandi CLI Personalizzati in Node.js

Impara a creare applicazioni a riga di comando professionali: da script semplici a tool avanzati con argomenti, opzioni e colori

🎯 Cos'è una CLI?

Command Line Interface

Una CLI (Command Line Interface) è un'applicazione che si esegue dal terminale invece che con interfaccia grafica. Esempi famosi: git, npm, docker

Perché creare CLI?

  • ✅ Automazione di task ripetitivi
  • ✅ Tool di sviluppo per il team
  • ✅ Script di deployment e configurazione
  • ✅ Utility per gestire dati (import/export, backup)
  • ✅ Perfetti per server senza GUI

🔧 Il metodo base: process.argv

process.argv è un array che contiene tutti gli argomenti passati al comando. Documentazione: Node.js process.argv

Struttura di process.argv

// Comando: node app.js create user --name Mario --age 30

console.log(process.argv);
// Output:
[
  '/usr/bin/node',           // [0] Path di Node.js
  '/path/to/app.js',         // [1] Path dello script
  'create',                  // [2] Primo argomento
  'user',                    // [3] Secondo argomento
  '--name',                  // [4] Flag
  'Mario',                   // [5] Valore del flag
  '--age',                   // [6] Flag
  '30'                       // [7] Valore del flag
]

Esempio base: calcolatrice CLI

// calc.js - Uso: node calc.js add 5 3

const [,, operation, num1, num2] = process.argv;

const a = parseFloat(num1);
const b = parseFloat(num2);

if (isNaN(a) || isNaN(b)) {
    console.error('❌ Errore: Inserisci numeri validi');
    process.exit(1);
}

let result;

switch(operation) {
    case 'add':
        result = a + b;
        break;
    case 'subtract':
        result = a - b;
        break;
    case 'multiply':
        result = a * b;
        break;
    case 'divide':
        if (b === 0) {
            console.error('❌ Errore: Divisione per zero');
            process.exit(1);
        }
        result = a / b;
        break;
    default:
        console.error('❌ Operazione non supportata');
        console.log('Usa: add, subtract, multiply, divide');
        process.exit(1);
}

console.log(`✅ Risultato: ${result}`);

Test della calcolatrice

$ node calc.js add 10 5
✅ Risultato: 15

$ node calc.js multiply 4 3
✅ Risultato: 12

$ node calc.js divide 10 0
❌ Errore: Divisione per zero

⚙️ Parsing argomenti avanzato

Per CLI complesse, conviene parsare flag e opzioni in modo strutturato

Parser manuale semplice

// parseArgs.js - Utility per parsare argomenti

export function parseArgs(argv) {
    const args = {
        _: [],        // Argomenti posizionali
        flags: {}     // Flag nominati
    };
    
    for (let i = 2; i < argv.length; i++) {
        const arg = argv[i];
        
        if (arg.startsWith('--')) {
            // Flag lungo: --name=Mario o --name Mario
            const flagName = arg.slice(2);
            
            if (flagName.includes('=')) {
                const [key, value] = flagName.split('=');
                args.flags[key] = value;
            } else {
                // Valore nel prossimo argomento
                const nextArg = argv[i + 1];
                if (nextArg && !nextArg.startsWith('-')) {
                    args.flags[flagName] = nextArg;
                    i++; // Salta il prossimo
                } else {
                    args.flags[flagName] = true; // Flag booleano
                }
            }
        } else if (arg.startsWith('-')) {
            // Flag corto: -v
            const flagName = arg.slice(1);
            args.flags[flagName] = true;
        } else {
            // Argomento posizionale
            args._.push(arg);
        }
    }
    
    return args;
}

// Utilizzo
const args = parseArgs(process.argv);
console.log(args);

// Comando: node app.js create user --name Mario --age 30 -v
// Output: {
//   _: ['create', 'user'],
//   flags: { name: 'Mario', age: '30', v: true }
// }

Esempio: User Manager CLI

// user-cli.js
import { parseArgs } from './parseArgs.js';

const args = parseArgs(process.argv);
const [command, entity] = args._;

if (command === 'create' && entity === 'user') {
    const { name, email, age } = args.flags;
    
    if (!name || !email) {
        console.error('❌ --name e --email sono obbligatori');
        process.exit(1);
    }
    
    const user = {
        id: Date.now(),
        name,
        email,
        age: age ? parseInt(age) : null,
        createdAt: new Date()
    };
    
    console.log('✅ Utente creato:', user);
    
    // Qui salveresti su file/database
} else {
    console.log('Uso: node user-cli.js create user --name <nome> --email <email> [--age <eta>]');
}

// Test:
// node user-cli.js create user --name Mario --email mario@example.com --age 30

📦 Librerie CLI professionali

Per CLI complesse, usa librerie specializzate che gestiscono parsing, validazione e help automatico

Commander.js - La più popolare

// Installazione
npm install commander

// app.js con Commander
import { program } from 'commander';

program
    .name('mytool')
    .description('Tool CLI per gestire utenti')
    .version('1.0.0');

program
    .command('create')
    .description('Crea un nuovo utente')
    .requiredOption('-n, --name <name>', 'Nome utente')
    .requiredOption('-e, --email <email>', 'Email utente')
    .option('-a, --age <age>', 'Età utente', parseInt)
    .option('-v, --verbose', 'Output dettagliato')
    .action((options) => {
        if (options.verbose) {
            console.log('Creazione utente con opzioni:', options);
        }
        
        const user = {
            name: options.name,
            email: options.email,
            age: options.age || null
        };
        
        console.log('✅ Utente creato:', user);
    });

program
    .command('list')
    .description('Lista tutti gli utenti')
    .option('-l, --limit <number>', 'Numero massimo risultati', 10)
    .action((options) => {
        console.log(`Mostro ${options.limit} utenti...`);
    });

program.parse();

// Help automatico:
// node app.js --help
// node app.js create --help

Yargs - Alternativa potente

// Installazione
npm install yargs

// app.js con Yargs
import yargs from 'yargs';
import { hideBin } from 'yargs/helpers';

yargs(hideBin(process.argv))
    .command('create <name>', 'Crea utente', (yargs) => {
        return yargs
            .positional('name', {
                describe: 'Nome utente',
                type: 'string'
            })
            .option('email', {
                alias: 'e',
                type: 'string',
                demandOption: true,
                describe: 'Email utente'
            })
            .option('age', {
                alias: 'a',
                type: 'number',
                describe: 'Età utente'
            });
    }, (argv) => {
        console.log('Creo utente:', argv.name, argv.email);
    })
    .command('delete <id>', 'Elimina utente', (yargs) => {
        return yargs.positional('id', {
            describe: 'ID utente',
            type: 'number'
        });
    }, (argv) => {
        console.log('Elimino utente:', argv.id);
    })
    .demandCommand(1, 'Devi specificare un comando')
    .help()
    .argv;
Libreria Pro Contro Usa quando
Commander Semplice, leggero, popolare Meno feature avanzate CLI medio-semplici
Yargs Molto potente, validazione built-in Più complesso, più pesante CLI complesse con tante opzioni
Manuale Zero dipendenze, massimo controllo Più codice da scrivere CLI semplicissime o learning

CLI completa con tutti gli extra

// user-tool.js - CLI professionale
import { program } from 'commander';
import chalk from 'chalk';
import ora from 'ora';
import inquirer from 'inquirer';
import fs from 'fs/promises';

program
    .name('user-tool')
    .description('Gestione utenti')
    .version('1.0.0');

program
    .command('create')
    .description('Crea nuovo utente')
    .option('-i, --interactive', 'Modalità interattiva')
    .action(async (options) => {
        let userData;
        
        if (options.interactive) {
            // Modalità interattiva
            userData = await inquirer.prompt([
                {
                    type: 'input',
                    name: 'name',
                    message: 'Nome:',
                    validate: (input) => input.length > 0
                },
                {
                    type: 'input',
                    name: 'email',
                    message: 'Email:',
                    validate: (input) => input.includes('@')
                },
                {
                    type: 'number',
                    name: 'age',
                    message: 'Età:'
                }
            ]);
        } else {
            console.error(chalk.red('Usa --interactive per modalità guidata'));
            return;
        }
        
        const spinner = ora('Creazione utente...').start();
        
        // Simula operazione async
        await new Promise(resolve => setTimeout(resolve, 1500));
        
        const user = {
            id: Date.now(),
            ...userData,
            createdAt: new Date()
        };
        
        // Salva su file
        try {
            const users = JSON.parse(await fs.readFile('users.json', 'utf-8'));
            users.push(user);
            await fs.writeFile('users.json', JSON.stringify(users, null, 2));
            
            spinner.succeed(chalk.green('Utente creato con successo!'));
            console.log(chalk.gray(JSON.stringify(user, null, 2)));
        } catch (error) {
            spinner.fail(chalk.red('Errore durante il salvataggio'));
            console.error(error.message);
        }
    });

program.parse();

🔐 Rendere il comando globale

Puoi installare la tua CLI globalmente come npm o git

1. Aggiungi shebang

#!/usr/bin/env node

// Il resto del codice...
import { program } from 'commander';

2. Configura package.json

{
  "name": "my-cli-tool",
  "version": "1.0.0",
  "type": "module",
  "bin": {
    "mytool": "./bin/cli.js"
  },
  "files": [
    "bin",
    "lib"
  ]
}

3. Rendi eseguibile

# Linux/Mac
chmod +x bin/cli.js

4. Installa globalmente

# Dalla cartella del progetto
npm link

# Ora puoi usare il comando ovunque
mytool create --name Mario

# Per disinstallare
npm unlink -g my-cli-tool

5. Pubblica su NPM (opzionale)

# Crea account su npmjs.com
npm login

# Pubblica il package
npm publish

# Altri possono installare con
npm install -g my-cli-tool

💡 Best Practices per CLI

✅ Help sempre disponibile

Implementa --help e -h per ogni comando. L'utente deve capire come usare il tool senza documentazione esterna.

📝 Output chiaro

Usa emoji e colori per distinguere successo (✅ verde), errori (❌ rosso), warning (⚠️ giallo), info (ℹ️ blu).

🔄 Feedback progressivo

Per operazioni lunghe usa spinner o progress bar. L'utente deve sapere che qualcosa sta accadendo.

⚠️ Gestione errori

Mostra messaggi di errore utili, non stack trace criptici. Suggerisci cosa fare per risolvere.

🔐 Validazione input

Valida SEMPRE l'input utente. Non dare per scontato che sia corretto.

📤 Exit code corretti

process.exit(0) per successo, process.exit(1) per errori. Permette l'uso in script automatici.

🎓 Progetto completo: Todo CLI

Mettiamo tutto insieme in una CLI funzionale completa

// todo-cli.js
#!/usr/bin/env node

import { program } from 'commander';
import chalk from 'chalk';
import fs from 'fs/promises';
import path from 'path';

const DATA_FILE = path.join(process.cwd(), 'todos.json');

async function loadTodos() {
    try {
        const data = await fs.readFile(DATA_FILE, 'utf-8');
        return JSON.parse(data);
    } catch {
        return [];
    }
}

async function saveTodos(todos) {
    await fs.writeFile(DATA_FILE, JSON.stringify(todos, null, 2));
}

program
    .name('todo')
    .description('Gestione TODO da terminale')
    .version('1.0.0');

program
    .command('add <task>')
    .description('Aggiungi un nuovo task')
    .option('-p, --priority <level>', 'Priorità (low|medium|high)', 'medium')
    .action(async (task, options) => {
        const todos = await loadTodos();
        const newTodo = {
            id: Date.now(),
            task,
            priority: options.priority,
            completed: false,
            createdAt: new Date()
        };
        
        todos.push(newTodo);
        await saveTodos(todos);
        
        console.log(chalk.green('✅ Task aggiunto:'), task);
    });

program
    .command('list')
    .description('Lista tutti i task')
    .option('-a, --all', 'Mostra anche completati')
    .action(async (options) => {
        const todos = await loadTodos();
        let filtered = todos;
        
        if (!options.all) {
            filtered = todos.filter(t => !t.completed);
        }
        
        if (filtered.length === 0) {
            console.log(chalk.gray('Nessun task da mostrare'));
            return;
        }
        
        console.log(chalk.bold('\n📋 TODO LIST\n'));
        filtered.forEach(todo => {
            const status = todo.completed ? chalk.green('✓') : chalk.gray('○');
            const priority = todo.priority === 'high' ? chalk.red('[!]') : 
                           todo.priority === 'medium' ? chalk.yellow('[·]') : 
                           chalk.gray('[·]');
            const text = todo.completed ? chalk.gray.strikethrough(todo.task) : todo.task;
            
            console.log(`${status} ${priority} ${text} ${chalk.gray('#' + todo.id)}`);
        });
        console.log();
    });

program
    .command('done <id>')
    .description('Segna task come completato')
    .action(async (id) => {
        const todos = await loadTodos();
        const todo = todos.find(t => t.id === parseInt(id));
        
        if (!todo) {
            console.log(chalk.red('❌ Task non trovato'));
            return;
        }
        
        todo.completed = true;
        await saveTodos(todos);
        
        console.log(chalk.green('✅ Task completato:'), todo.task);
    });

program
    .command('remove <id>')
    .description('Rimuovi un task')
    .action(async (id) => {
        let todos = await loadTodos();
        const index = todos.findIndex(t => t.id === parseInt(id));
        
        if (index === -1) {
            console.log(chalk.red('❌ Task non trovato'));
            return;
        }
        
        const removed = todos.splice(index, 1)[0];
        await saveTodos(todos);
        
        console.log(chalk.yellow('🗑️  Task rimosso:'), removed.task);
    });

program.parse();

Utilizzo

# Aggiungi task
todo add "Studiare Node.js" -p high
todo add "Fare la spesa"

# Lista task
todo list
todo list --all

# Completa task
todo done 1234567890

# Rimuovi task
todo remove 1234567890

# Help
todo --help
todo add --help
